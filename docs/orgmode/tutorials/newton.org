* Newton

Shifted Newton needs a Hessian. On a convex quadratic it takes the Newton step ~(H)^{-1} g~. The oracle is ~HessianOracle::unbounded~.

The program is ~examples/tutorial_newton.rs~. Continuous integration runs it.

#+begin_src rust
use ndarray::array;
use rgmin::{Control, HessianOracle, NewtonKind, minimize_newton};

let value = |x: ndarray::ArrayView1<f64>| {
    let f = 0.5 * (4.0 * x[0] * x[0] + x[1] * x[1]) - x[0] + 2.0 * x[1];
    (f, array![4.0 * x[0] - 1.0, x[1] + 2.0])
};
let hess = |_x: ndarray::ArrayView1<f64>| array![[4.0, 0.0], [0.0, 1.0]];
let obj = HessianOracle::unbounded(2, value, hess);
let report = minimize_newton(&obj, array![0.0, 0.0], &Control::default(), NewtonKind::Shifted).unwrap();
assert!((report.coords[0] - 0.25).abs() < 1e-6);
#+end_src
